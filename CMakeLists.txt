    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
    add_compile_options(/source-charset:utf-8 /execution-charset:.949)

    add_compile_options(-Wall -Wextra -Werror=implicit-function-declaration -fsigned-char)
    if(RAGE_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined,float-cast-overflow)
    endif()
endif()

# A platform module is one header with one file per implementation: <module>_win32.c on Windows,
# <module>_posix.c on Linux, macOS and Android.
if(WIN32)
    set(RAGE_PLATFORM_FILES win32)
else()
    set(RAGE_PLATFORM_FILES posix)
endif()
# core/utf8.c is the client's text encoding, which the code-page conversions use for UTF-8.
add_library(rage_platform STATIC src/core/utf8.c)
foreach(module clock code_page filesystem net settings sha256 thread)
    target_sources(rage_platform PRIVATE src/platform/${module}_${RAGE_PLATFORM_FILES}.c)
endforeach()
target_include_directories(rage_platform PUBLIC src)
if(WIN32)
    # The settings store keeps UTF-8 text in the registry through platform/win_utf8.c's W calls.
    target_sources(rage_platform PRIVATE src/platform/win_utf8.c)
    target_link_libraries(rage_platform PUBLIC winmm ws2_32 user32 advapi32)
else()
    find_package(Threads REQUIRED)
    target_compile_definitions(rage_platform PRIVATE _POSIX_C_SOURCE=200809L)
    if(APPLE)
        target_compile_definitions(rage_platform PRIVATE _DARWIN_C_SOURCE)
    endif()
    target_link_libraries(rage_platform PUBLIC Threads::Threads)
    if(ANDROID)
        target_sources(rage_platform PRIVATE src/platform/code_page_iconv.c src/platform/code_page_tables.c)
    else()
        find_package(Iconv REQUIRED)
        target_link_libraries(rage_platform PUBLIC Iconv::Iconv)
    endif()
endif()

function(rage_enable_assertions target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()

function(rage_test name)
    add_executable(test_${name} tests/test_${name}.c ${ARGN})
    target_include_directories(test_${name} PRIVATE src)
    rage_enable_assertions(test_${name})
    if(NOT MSVC)
        target_link_libraries(test_${name} PRIVATE m)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
endfunction()
